cmake_minimum_required(VERSION 3.25)
project(qwen_asr_mlx VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
set(QWEN_ASR_BACKEND "metal" CACHE STRING "metal, cuda or cpu (must match installed MLX)")
set_property(CACHE QWEN_ASR_BACKEND PROPERTY STRINGS metal cuda cpu)
if(NOT QWEN_ASR_BACKEND MATCHES "^(metal|cuda|cpu)$")
  message(FATAL_ERROR "QWEN_ASR_BACKEND must be metal, cuda or cpu")
endif()
if(QWEN_ASR_BACKEND STREQUAL "metal" AND NOT APPLE)
  message(FATAL_ERROR "Metal requires macOS; choose cuda or cpu on Linux")
endif()
find_package(MLX 0.32.4 EXACT CONFIG REQUIRED)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
add_library(qwen_asr_mlx SHARED model.cpp vendor/qwen-asr/qwen_asr_tokenizer.c)
target_include_directories(qwen_asr_mlx PRIVATE vendor/qwen-asr PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(qwen_asr_mlx PRIVATE mlx nlohmann_json::nlohmann_json)
if(QWEN_ASR_BACKEND STREQUAL "metal")
  target_compile_definitions(qwen_asr_mlx PRIVATE QWEN_ASR_METAL=1)
elseif(QWEN_ASR_BACKEND STREQUAL "cuda")
  target_compile_definitions(qwen_asr_mlx PRIVATE QWEN_ASR_CUDA=1)
endif()
# Find the installed MLX dylib/so after installing this shared library.
set_target_properties(qwen_asr_mlx PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS qwen_asr_mlx LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/qwen_asr_mlx.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_file(qwen-asr-mlx.pc.in qwen-asr-mlx.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/qwen-asr-mlx.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
